set(NGOSEN_SETTINGS_DIR "${NGOSEN_IBUS_DATADIR}/settings")
set(NGOSEN_LOCALEDIR "${CMAKE_INSTALL_FULL_LOCALEDIR}")
set(NGOSEN_FONT "${NGOSEN_IBUS_DATADIR}/fonts/XanhMono-Regular.ttf")
foreach(path NGOSEN_IBUS_EXEC NGOSEN_SETUP_EXEC NGOSEN_IBUS_DATADIR)
    if (NOT ${path})
        message(FATAL_ERROR "${path} must be set before settings/")
    endif()
endforeach()
configure_file(ibus-setup-ngosen.in ${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-ngosen @ONLY)
configure_file(ibus-setup-ngosen.desktop.in ${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-ngosen.desktop @ONLY)

install(PROGRAMS ${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-ngosen DESTINATION ${CMAKE_INSTALL_LIBEXECDIR})
install(DIRECTORY ngosen_settings DESTINATION ${NGOSEN_SETTINGS_DIR} FILES_MATCHING PATTERN "*.py")
# GNOME Settings opens an IBus engine's settings through ibus-setup-<engine>.desktop.
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-ngosen.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES io.github.ngosen.IBusNgoSen.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
install(FILES ${PROJECT_SOURCE_DIR}/ngosen/settings-gui/fonts/XanhMono-Regular.ttf
              ${PROJECT_SOURCE_DIR}/ngosen/LICENSES/OFL-1.1.txt
        DESTINATION ${NGOSEN_IBUS_DATADIR}/fonts)

# The window tests need a display; without xvfb-run they are skipped.
find_program(XVFB_RUN xvfb-run)
set(display_runner)
if (XVFB_RUN)
    set(display_runner ${XVFB_RUN} -a)
endif()
add_test(NAME settings_window COMMAND ${display_runner} ${Python3_EXECUTABLE} -m unittest discover -s tests -t . -v
         WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
# GDK_BACKEND keeps GTK on the Xvfb display when the desktop session is Wayland.
set_tests_properties(settings_window PROPERTIES ENVIRONMENT "NGOSEN_IBUS_ENGINE=$<TARGET_FILE:ibus-engine-ngosen>;GDK_BACKEND=x11")

find_program(DESKTOP_FILE_VALIDATE desktop-file-validate)
if (DESKTOP_FILE_VALIDATE)
    add_test(NAME desktop_file COMMAND ${DESKTOP_FILE_VALIDATE} ${CMAKE_CURRENT_BINARY_DIR}/ibus-setup-ngosen.desktop)
endif()
find_program(APPSTREAMCLI appstreamcli)
if (APPSTREAMCLI)
    add_test(NAME metainfo COMMAND ${APPSTREAMCLI} validate --no-net ${CMAKE_CURRENT_SOURCE_DIR}/io.github.ngosen.IBusNgoSen.metainfo.xml)
endif()
